using UnityEngine;

public class SpeedBoost : MonoBehaviour
{
    private bool boostActive = false;
    private float timer = 1f;

    private void FixedUpdate()
    {
        if (boostActive)
        {
            if (timer > 0f)
            {
                Player.AddForce(speedBoostMultiplier * Player.forward, "Force");
                ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
                Player.CustomizeTrail(0f, 0f, new Color(1f, 1f, 1f, 1f));
                timer = timer - null.deltaTime;
            }
            else
            {
                boostActive = false;
                Player.ResetTrail();
                ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
            }
            boostActive = false;
            Player.ResetTrail();
            ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
        }
        if (timer > 0f)
        {
            Player.AddForce(speedBoostMultiplier * Player.forward, "Force");
            ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
            Player.CustomizeTrail(0f, 0f, new Color(1f, 1f, 1f, 1f));
            timer = timer - null.deltaTime;
        }
        else
        {
            boostActive = false;
            Player.ResetTrail();
            ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
        }
        boostActive = false;
        Player.ResetTrail();
        ShipShader.SetColor("", new Color(1f, 1f, 1f, 1f));
    }

    private void OnTriggerEnter()
    {
        boostActive = true;
        timer = 1f;
    }
}
