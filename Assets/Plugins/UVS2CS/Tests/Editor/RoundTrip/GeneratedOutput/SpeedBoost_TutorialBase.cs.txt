using UnityEngine;

public class SpeedBoost_TutorialBase : MonoBehaviour
{
    private bool boostActive = false;
    private float timer = 1f;

    private void FixedUpdate()
    {
        if (boostActive)
        {
            if (timer > 0f)
            {
                Player.AddForce(speedBoostMultiplier * Player.forward, "Impulse");
                ShipShader.SetColor("_EmissionColor", new Color(1f, 0.7886922f, 0f, 1f));
            }
            else
            {
                boostActive = false;
            }
            Player.AddForce(speedBoostMultiplier * Player.forward, "Impulse");
            ShipShader.SetColor("_EmissionColor", new Color(1f, 0.7886922f, 0f, 1f));
        }
        if (timer > 0f)
        {
            Player.AddForce(speedBoostMultiplier * Player.forward, "Impulse");
            ShipShader.SetColor("_EmissionColor", new Color(1f, 0.7886922f, 0f, 1f));
        }
        else
        {
            boostActive = false;
        }
        Player.AddForce(speedBoostMultiplier * Player.forward, "Impulse");
        ShipShader.SetColor("_EmissionColor", new Color(1f, 0.7886922f, 0f, 1f));
    }

    private void OnTriggerEnter()
    {
        boostActive = true;
        timer = 1f;
    }
}
