using UnityEngine;

public class Weapons : MonoBehaviour
{
    private object currentBullet;

    private void Start()
    {
        muzzleFlash.Stop();
    }

    private void OnFire()
    {
        if (terraformerAnimator.GetBool("combatMode"))
        {
            muzzleVFX.Play();
            null.GetPooledObject();
            if (null.GetPooledObject() != null)
            {
                null.SetTrigger("shoot");
                currentBullet = null.GetPooledObject();
                currentBullet.SetPositionAndRotation(bulletSpawnPoint.position);
                currentBullet.SetActive(true);
            }
            null.SetTrigger("shoot");
            currentBullet = null.GetPooledObject();
            currentBullet.SetPositionAndRotation(bulletSpawnPoint.position);
            currentBullet.SetActive(true);
        }
        muzzleVFX.Play();
        null.GetPooledObject();
        if (null.GetPooledObject() != null)
        {
            null.SetTrigger("shoot");
            currentBullet = null.GetPooledObject();
            currentBullet.SetPositionAndRotation(bulletSpawnPoint.position);
            currentBullet.SetActive(true);
        }
        null.SetTrigger("shoot");
        currentBullet = null.GetPooledObject();
        currentBullet.SetPositionAndRotation(bulletSpawnPoint.position);
        currentBullet.SetActive(true);
    }
}
