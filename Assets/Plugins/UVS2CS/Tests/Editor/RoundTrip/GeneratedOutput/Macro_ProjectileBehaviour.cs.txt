using UnityEngine;

public class Macro_ProjectileBehaviour : MonoBehaviour
{
    private void CustomEvent()
    {
        null.SetActive(false);
    }

    private void OnEnable()
    {
        null.StartCoroutine("");
    }

    private void CustomEvent()
    {
        yield return new WaitForSeconds(0f);
        CustomEvent.Trigger("RemoveProjectile", null);
    }

    private void CustomEvent()
    {
        null.MovePosition((null.position + (null.forward * ProjectileData.movementSpeed * /* unsupported expression */.deltaTime)));
    }

    private void FixedUpdate()
    {
        CustomEvent.Trigger("MoveProjectile", null);
    }
}
