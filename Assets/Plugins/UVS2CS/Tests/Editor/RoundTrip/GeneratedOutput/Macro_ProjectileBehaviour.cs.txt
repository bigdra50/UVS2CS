using UnityEngine;

public class Macro_ProjectileBehaviour : MonoBehaviour
{
    private void OnMoveProjectile()
    {
        null.MovePosition(null + null);
    }

    private void OnTimerToAutoRemoveProjectile()
    {
        yield return new WaitForSeconds(null.autoRemoveCountdown);
        CustomEvent.Trigger("RemoveProjectile", null);
    }

    private void FixedUpdate()
    {
        CustomEvent.Trigger("MoveProjectile", null);
    }

    private void OnEnable()
    {
        null.StartCoroutine("TimerToAutoRemoveProjectile");
    }

    private void OnRemoveProjectile()
    {
        null.SetActive(false);
    }
}
