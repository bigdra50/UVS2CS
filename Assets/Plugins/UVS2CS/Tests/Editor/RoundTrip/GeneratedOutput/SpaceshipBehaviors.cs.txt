using UnityEngine;

public class SpaceshipBehaviors : MonoBehaviour
{
    private float rateOfFire = 0.8f;

    private void FixedUpdate()
    {
        if (data.thrustInput > 0.1f)
        {
            spaceshipRigidbody.velocity = (null.forward * data.thrustAmount * Max(data.thrustInput, 0.4f));
        }
        spaceshipRigidbody.velocity = (null.forward * data.thrustAmount * Max(data.thrustInput, 0.4f));
    }

    private void OnboostVisual()
    {
        null.boostEffect.main.startColor = CustomEvent;
    }

    private void FixedUpdate_1()
    {
        Vector3..ctor(data.pitchSpeed * null.x, (null.z * -1f * data.yawSpeed), 0f);
        spaceshipRigidbody.AddRelativeTorque(.ctor(data.pitchSpeed * null.x, (null.z * -1f * data.yawSpeed), 0f));
        Quaternion.Slerp(0.5f);
        spaceshipRigidbody.rotation = Slerp(0.5f);
    }

    private void FixedUpdate_2()
    {
        if (data.shootInput)
        {
            null.PlayOneShot(PickRandomAudioClip);
        }
        null.PlayOneShot(PickRandomAudioClip);
    }
}
