using UnityEngine;

public class ProjectileBehavior : MonoBehaviour
{
    private void OnEnable()
    {
        /* Timer: complex timer */;
        CustomEvent.Trigger("RemoveProjectile", null);
    }

    private void OnMoveProjectile()
    {
        null.MovePosition(null + null);
    }

    private void OnRemoveProjectile()
    {
        null.SetActive(false);
    }

    private void OnCollisionRaycastCheck()
    {
        if (Raycast(null.position, null.forward, RaycastHit.distance, 100))
        {
            if (RaycastHit.distance <= 0.collisionRayDistance)
            {
                CustomEvent.Trigger("RemoveProjectile", null);
            }
            CustomEvent.Trigger("RemoveProjectile", null);
        }
        if (RaycastHit.distance <= 0.collisionRayDistance)
        {
            CustomEvent.Trigger("RemoveProjectile", null);
        }
        CustomEvent.Trigger("RemoveProjectile", null);
    }

    private void FixedUpdate()
    {
        CustomEvent.Trigger("MoveProjectile", null);
        CustomEvent.Trigger("CollisionRaycastCheck", null);
    }
}
